#==============================================================================
# Simple scene for ray tracer testing
# Start with one object, then add more as testing progresses
#==============================================================================

#all the instances
instance box models/box-outside.obj
instance sphere models/sphere.obj

#------------------------------------------------------------------------------
# Materials
#------------------------------------------------------------------------------

# Red material for the box
material red-material
emission 0 0 0
ambient 0.2 0 0
diffuse 0.8 0 0
specular 1 1 1
shininess 50
end-material

# Blue material for the sphere
material blue-material
emission 0 0 0
ambient 0 0 0.2
diffuse 0 0 0.8
specular 1 1 1
shininess 100
absorption 0.5
reflection 0.5
end-material

# White material for the floor
material white-material
emission 0 0 0
ambient 0.2 0.2 0.2
diffuse 0.8 0.8 0.8
specular 0.5 0.5 0.5
shininess 20
end-material

#------------------------------------------------------------------------------
# Lights
#------------------------------------------------------------------------------

# Main directional-like light from above-right
light main-light
ambient 0.2 0.2 0.2
diffuse 0.7 0.7 0.7
specular 0.8 0.8 0.8
position 50 100 50
end-light

# Fill light from left
light fill-light
ambient 0.1 0.1 0.1
diffuse 0.3 0.3 0.3
specular 0.2 0.2 0.2
position -50 50 50
end-light

#------------------------------------------------------------------------------
# Textures (using white for now, no texture mapping needed)
#------------------------------------------------------------------------------
image white textures/white.ppm

#------------------------------------------------------------------------------
# Scene Graph
#------------------------------------------------------------------------------

# Root node
group root root
assign-light main-light root
assign-light fill-light root

#------------------------------------------------------------------------------
# Object 1: Red box on the left
#------------------------------------------------------------------------------
translate box-translate box-translate -15 0 0
scale box-scale box-scale 8 8 8
leaf red-box red-box instanceof box 
assign-material red-box red-material
assign-texture red-box white

add-child red-box box-scale
add-child box-scale box-translate
add-child box-translate root

#------------------------------------------------------------------------------
# Object 2: Blue sphere on the right
#------------------------------------------------------------------------------
translate sphere-translate sphere-translate 15 0 0
scale sphere-scale sphere-scale 10 10 10
leaf blue-sphere blue-sphere instanceof sphere 
assign-material blue-sphere blue-material
assign-texture blue-sphere white

add-child blue-sphere sphere-scale
add-child sphere-scale sphere-translate
add-child sphere-translate root

#------------------------------------------------------------------------------
# Object 3: White floor (ground plane)
#------------------------------------------------------------------------------
translate floor-translate floor-translate 0 -15 0
scale floor-scale floor-scale 100 2 100
leaf floor floor instanceof box 
assign-material floor white-material
assign-texture floor white

add-child floor floor-scale
add-child floor-scale floor-translate
add-child floor-translate root

assign-root root
