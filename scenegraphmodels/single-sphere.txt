#==============================================================================
# Minimal scene for ray tracer debugging
# Only ONE object - easiest to debug intersection code
#==============================================================================

#all the instances
instance sphere models/sphere.obj

#------------------------------------------------------------------------------
# Material
#------------------------------------------------------------------------------

# Simple white material
material white-material
emission 0 0 0
ambient 0.2 0.2 0.2
diffuse 0.8 0.8 0.8
specular 1 1 1
shininess 50
end-material

#------------------------------------------------------------------------------
# Light
#------------------------------------------------------------------------------

# Single light from front-right-top
light main-light
ambient 0.3 0.3 0.3
diffuse 0.7 0.7 0.7
specular 0.8 0.8 0.8
position 30 30 30
end-light

#------------------------------------------------------------------------------
# Texture
#------------------------------------------------------------------------------
image white textures/white.ppm

#------------------------------------------------------------------------------
# Scene Graph - Just one sphere at origin
#------------------------------------------------------------------------------

group root root
assign-light main-light root

# Single sphere, scaled to radius 10, at origin
scale sphere-scale sphere-scale 10 10 10
leaf one-sphere one-sphere instanceof sphere 
assign-material one-sphere white-material
assign-texture one-sphere white

add-child one-sphere sphere-scale
add-child sphere-scale root

assign-root root
