#==============================================================================
# Minimal scene for ray tracer debugging - Single Box
# Use this to test box intersection before moving to sphere
#==============================================================================

#all the instances
instance box models/box-outside.obj

#------------------------------------------------------------------------------
# Material
#------------------------------------------------------------------------------

# Simple red material (easy to see)
material red-material
emission 0 0 0
ambient 0.2 0 0
diffuse 0.8 0 0
specular 1 1 1
shininess 50
end-material

#------------------------------------------------------------------------------
# Light
#------------------------------------------------------------------------------

# Single light from front-right-top
light main-light
ambient 0.3 0.3 0.3
diffuse 0.7 0.7 0.7
specular 0.8 0.8 0.8
position 30 30 30
end-light

#------------------------------------------------------------------------------
# Texture
#------------------------------------------------------------------------------
image white textures/white.ppm

#------------------------------------------------------------------------------
# Scene Graph - Just one box at origin
#------------------------------------------------------------------------------

group root root
assign-light main-light root

# Single box, scaled to size 10x10x10, at origin
scale box-scale box-scale 10 10 10
leaf one-box one-box instanceof box 
assign-material one-box red-material
assign-texture one-box white

add-child one-box box-scale
add-child box-scale root

assign-root root
